Running the programs

# Compile & run a single file (Java 11+ can even skip javac):
javac Calculator.java && java Calculator
java Calculator.java

# Enable assertions
java -ea AssertDemo

# JavaFX (download the OpenJFX SDK)
javac --module-path /path/to/javafx-sdk/lib --add-modules javafx.controls KeyEventDemo.java
java  --module-path /path/to/javafx-sdk/lib --add-modules javafx.controls KeyEventDemo

# JDBC with MySQL / SQLite driver
java -cp .:mysql-connector-j.jar JdbcDemo
java -cp .:sqlite-jdbc.jar JdbcDemo        # URL "jdbc:sqlite:college.db" - no server needed
